function onOpen() { SpreadsheetApp.getUi().createMenu('Wings Tools').addItem('🧹 Hapus Data Double', 'removeDuplicates').addToUi(); } function removeDuplicates() { var ss = SpreadsheetApp.getActiveSpreadsheet(), sheetsToClean = ['Bin Mat', 'Available stock'], totalRemoved = 0; sheetsToClean.forEach(function(sheetName) { var sheet = ss.getSheetByName(sheetName); if (sheet) { var data = sheet.getDataRange().getValues(), newData = [data[0]], seen = {}; for (var i = 1; i < data.length; i++) { var key = String(data[i][0]).trim() + "_" + String(data[i][1]).trim() + "_" + String(data[i][2]).trim(); if (key === "__") continue; if (!seen[key]) { seen[key] = true; newData.push(data[i]); } else { totalRemoved++; } } if (newData.length < data.length) { sheet.clearContents(); sheet.getRange(1, 1, newData.length, newData[0].length).setValues(newData); } } }); SpreadsheetApp.getUi().alert("Optimalisasi Selesai!\n" + totalRemoved + " baris data duplikat berhasil dihapus."); } function doGet(e) { return HtmlService.createHtmlOutputFromFile('Index').setTitle('WMS Bin Checker').setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL).addMetaTag('viewport', 'width=device-width, initial-scale=1'); } function doPost(e) { try { var payload = JSON.parse(e.postData.contents); return ContentService.createTextOutput(JSON.stringify(handleWebAppRequest(payload))).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } } function handleWebAppRequest(payload) { if (typeof payload === 'string') payload = JSON.parse(payload); if (payload.action === "init") return getInitData(); if (payload.action === "checkBin") return checkBinLogic(payload.bin); if (payload.action === "submitBinAudit") return saveBinAudit(payload.bin, payload.auditData); if (payload.action === "checkAvailableStock") return checkAvailableStock(payload.bin, payload.mid); // UPDATE: Menerima parameter actionMode if (payload.action === "addNewBin") return addNewBinToStock(payload.bin, payload.mid, payload.isEmpty, payload.emptyRows, payload.actionMode); } // ========================================================================= // UPDATE: DASHBOARD PROGRES PICK SLOT & REVERSE BIN + DATA TABEL // ========================================================================= function getInitData() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var binMatSheet = ss.getSheetByName('Bin Mat'), stockSheet = ss.getSheetByName('Available stock'), masterSheet = ss.getSheetByName('Master MID'); var binList = [], masterMids = {}; if (masterSheet) { var mData = masterSheet.getDataRange().getValues(); for (var i = 1; i < mData.length; i++) { var mId = String(mData[i][0]).trim(), mName = String(mData[i][1]).trim(); if (mId !== "") masterMids[mId] = mName; } } var totalBinMat = 0, binMat0102InStock = 0, binMat0102NotInStock = 0, stock0102NotInBinMat = 0; var stockMap = {}, stockBins0102 = []; var reverseBin = { target: 0, match: 0, audited: 0, kosong: 0, belum: 0, listBelum: [], listNotMatch: [] }; if (stockSheet) { var stockData = stockSheet.getDataRange().getValues(); for (var i = 1; i < stockData.length; i++) { var sBin = String(stockData[i][0]).trim().toUpperCase(), sProd = String(stockData[i][1]).trim(), sMid = String(stockData[i][2]).trim(); var sMatch = String(stockData[i][3]).trim().toLowerCase() === 'v', sNotMatch = String(stockData[i][4]).trim().toLowerCase() === 'v'; var sKosong = String(stockData[i][5]).trim().toUpperCase() === 'ITEM KOSONG' || sProd === '-' || sProd === ''; if (sBin !== "" || sProd !== "") { stockMap[sBin + "_" + sProd.toUpperCase()] = true; var suffix = sBin.slice(-2); if (suffix === '01' || suffix === '02') stockBins0102.push(sBin + "_" + sProd.toUpperCase()); if (sBin && binList.indexOf(sBin) === -1) binList.push(sBin); if (suffix === '03' || suffix === '04' || suffix === '05') { if (sProd !== "") { reverseBin.target++; if (sMatch) reverseBin.match++; if (sKosong) reverseBin.kosong++; if (sMatch || sNotMatch || sKosong) { reverseBin.audited++; if (sNotMatch) reverseBin.listNotMatch.push({bin: sBin, product: sProd, mid: sMid}); } else { reverseBin.belum++; reverseBin.listBelum.push({bin: sBin, product: sProd, mid: sMid}); } } } } } } var binMatMap = {}; var pickSlot = { target: 0, match: 0, audited: 0, kosong: 0, belum: 0, listBelum: [], listNotMatch: [] }; if (binMatSheet) { var data = binMatSheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { var b = String(data[i][0]).trim().toUpperCase(), p = String(data[i][1]).trim(), m = String(data[i][2]).trim(); if (b === "" && p === "" && m === "") continue; totalBinMat++; if (b && binList.indexOf(b) === -1) binList.push(b); if (b && p) { binMatMap[b + "_" + p.toUpperCase()] = true; var bSuffix = b.slice(-2); if (bSuffix === '01' || bSuffix === '02') { if (stockMap[b + "_" + p.toUpperCase()]) binMat0102InStock++; else binMat0102NotInStock++; } } var isMatch = String(data[i][3]).trim().toLowerCase() === 'v', isNotMatch = String(data[i][4]).trim().toLowerCase() === 'v'; // LOGIKA KOSONG DIPERKUAT: Kolom F "ITEM KOSONG" atau Kolom B kosong/strip var isKosong = String(data[i][5]).trim().toUpperCase() === 'ITEM KOSONG' || p === '-' || p === ''; if (p !== "") { pickSlot.target++; if (isMatch) pickSlot.match++; if (isKosong) pickSlot.kosong++; if (isMatch || isNotMatch || isKosong) { pickSlot.audited++; if (isNotMatch) pickSlot.listNotMatch.push({bin: b, product: p, mid: m}); } else { pickSlot.belum++; pickSlot.listBelum.push({bin: b, product: p, mid: m}); } } } } for (var k = 0; k < stockBins0102.length; k++) { if (!binMatMap[stockBins0102[k]]) stock0102NotInBinMat++; } pickSlot.pctMatch = pickSlot.target > 0 ? ((pickSlot.match / pickSlot.target) * 100).toFixed(2) : 0; pickSlot.pctBelum = pickSlot.target > 0 ? ((pickSlot.belum / pickSlot.target) * 100).toFixed(2) : 0; reverseBin.pctMatch = reverseBin.target > 0 ? ((reverseBin.match / reverseBin.target) * 100).toFixed(2) : 0; reverseBin.pctBelum = reverseBin.target > 0 ? ((reverseBin.belum / reverseBin.target) * 100).toFixed(2) : 0; return { status: 'success', bins: binList, pickSlot: pickSlot, reverseBin: reverseBin, totalBinMat: totalBinMat, binMat0102InStock: binMat0102InStock, binMat0102NotInStock: binMat0102NotInStock, stock0102NotInBinMat: stock0102NotInBinMat, masterMids: masterMids }; } // ========================================================================= // UPDATE LOGIKA: MENGIRIM DATA PRODUK UNTUK POP-UP BIN KOSONG // ========================================================================= function checkBinLogic(bin) { var suffix = String(bin).trim().slice(-2), ss = SpreadsheetApp.getActiveSpreadsheet(); var targetSheetName = (suffix === '01' || suffix === '02') ? 'Bin Mat' : (suffix === '03' || suffix === '04' || suffix === '05') ? 'Available stock' : ''; if (!targetSheetName) return { status: 'error', message: 'Kode Bin tidak valid.' }; var targetSheet = ss.getSheetByName(targetSheetName); if (!targetSheet) return { status: 'error', message: 'Sheet tidak ditemukan.' }; var data = targetSheet.getDataRange().getValues(), products = [], auditedCount = 0, seenMap = {}; var scannedSummary = { match: [], notmatch: [], empty: [], unaudited: [] }; for (var i = 1; i < data.length; i++) { if (String(data[i][0]).trim().toUpperCase() === bin.toUpperCase()) { var prodName = String(data[i][1]).trim(), midVal = String(data[i][2]).trim(); var matchCol = String(data[i][3]).trim().toLowerCase(), notMatchCol = String(data[i][4]).trim().toLowerCase(), ketCol = String(data[i][5]).trim().toUpperCase(); var key = prodName + "_" + midVal; if (!seenMap[key]) { seenMap[key] = true; // Deteksi status saat ini dari Google Sheet var currentStatus = null; if (matchCol === 'v') currentStatus = 'match'; else if (notMatchCol === 'v') currentStatus = 'notmatch'; else if (ketCol === 'ITEM KOSONG') currentStatus = 'empty'; if (currentStatus !== null) auditedCount++; products.push({ row: i + 1, product: prodName, mid: midVal, status: currentStatus, remark: ketCol }); if (currentStatus !== null) { var details = { name: prodName, mid: midVal, remark: ketCol }; if (currentStatus === 'match') scannedSummary.match.push(details); else if (currentStatus === 'empty') scannedSummary.empty.push(details); else scannedSummary.notmatch.push(details); } else { scannedSummary.unaudited.push({ name: prodName, mid: midVal, remark: '' }); } } } } if (products.length === 0) { if (suffix === '03' || suffix === '04' || suffix === '05') return { status: 'bin_not_found_030405', bin: bin, suffix: suffix }; return { status: 'error', message: 'Bin kosong atau tidak ditemukan.' }; } // LOGIKA BARU: Hanya muncul pop-up 'Sudah Diaudit' jika JUMLAH AUDIT = JUMLAH TOTAL PRODUK DI BIN ITU var isFullyAudited = (products.length > 0 && auditedCount === products.length); if (isFullyAudited) return { status: 'already_scanned', bin: bin, summary: scannedSummary, products: products, suffix: suffix }; // Jika belum full diaudit, tetap munculkan list untuk diaudit sisanya return { status: 'success', type: 'POPUP_LIST', products: products, suffix: suffix }; } function saveBinAudit(bin, auditData) { var ss = SpreadsheetApp.getActiveSpreadsheet(), suffix = String(bin).trim().slice(-2); var targetSheetName = (suffix === '03' || suffix === '04' || suffix === '05') ? 'Available stock' : 'Bin Mat'; var targetSheet = ss.getSheetByName(targetSheetName); if (!targetSheet) return { status: 'error', message: 'Sheet ' + targetSheetName + ' tidak ditemukan!' }; for (var i = 0; i < auditData.length; i++) { var item = auditData[i], rowData = []; if (item.status === 'match') { rowData = [['v', '', '']]; } else if (item.status === 'empty') { rowData = [['', '', 'ITEM KOSONG']]; } else { // LOGIKA KETERANGAN: "ganti mid pada bin" var remarkText = 'ganti mid pada bin'; if (item.remark && item.remark.trim() !== '') remarkText += ' (' + item.remark + ')'; rowData = [['', 'v', remarkText]]; } targetSheet.getRange(item.row, 4, 1, 3).setValues(rowData); } return { status: 'success', message: 'Audit Bin ' + bin + ' berhasil disimpan!', stats: getInitData() }; } function checkAvailableStock(bin, mid) { var ss = SpreadsheetApp.getActiveSpreadsheet(), stockSheet = ss.getSheetByName('Available stock'); if (!stockSheet) return { status: 'error', message: 'Sheet "Available stock" tidak ditemukan!' }; var data = stockSheet.getDataRange().getValues(), isMatch = false, targetMID = String(mid).trim().toUpperCase(); for (var i = 0; i < data.length; i++) { if (data[i].join("||").toUpperCase().indexOf(targetMID) !== -1) { isMatch = true; break; } } return { status: isMatch ? 'success' : 'error', message: (isMatch ? 'MATCH! MID ' : 'MISMATCH! MID ') + mid + (isMatch ? ' ditemukan.' : ' tidak ditemukan.'), stats: getInitData() }; } // ========================================================================= // UPDATE LOGIKA: MENERIMA DATA CHECKBOX BANYAK BARIS UNTUK BIN KOSONG // ========================================================================= function addNewBinToStock(bin, mid, isEmpty, emptyRows, actionMode) { var ss = SpreadsheetApp.getActiveSpreadsheet(), suffix = String(bin).trim().slice(-2); var targetSheetName = (suffix === '01' || suffix === '02') ? return { status: 'success', message: 'Data produk berhasil ditambahkan ke baris baru di Bin ' + newBin + '!', stats: getInitData() }; } else { // Mode Timpa / Replace var targetData = targetSheet.getDataRange().getValues(), rowIndex = -1; return { status: 'success', message: 'Data produk berhasil ditambahkan!', stats: getInitData() }; } } }
